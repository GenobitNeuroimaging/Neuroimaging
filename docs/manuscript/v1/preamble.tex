% =====================================================================
% preamble.tex — packages, bibliography and macros.
% This is the only file that should change if we move to a journal
% template; main.tex and sections/ stay the same.
% =====================================================================

% --- Encoding and fonts ----------------------------------------------
\usepackage[T1]{fontenc}
\usepackage{newtxtext}  % Times-like: narrower, suits two columns
\usepackage[american]{babel}
\usepackage{microtype}

% --- Layout ----------------------------------------------------------
\usepackage[margin=0.8in, columnsep=0.25in]{geometry}
\usepackage[small, compact]{titlesec}
\usepackage{balance}  % evens out the columns on the last page

% Title and abstract across both columns, body below them.
% #1: file with the abstract.
\newcommand{\makefrontmatter}[1]{%
  \twocolumn[\begin{@twocolumnfalse}
    \maketitle
    \input{#1}
    \vspace{1.5em}
  \end{@twocolumnfalse}]%
}

% --- Math, figures and tables ----------------------------------------
\usepackage{amsmath}
\usepackage{newtxmath}  % after amsmath; includes the AMS symbols
\usepackage{graphicx}
\graphicspath{{figures/}}
\usepackage{array}     % p{} columns with ragged-right (tables/)
\usepackage{booktabs}
\usepackage[font=small, labelfont=bf, labelsep=period]{caption}

% --- Bibliography (APA) ----------------------------------------------
% Single source of references for the whole repository (see #4).
% references.bib is exported by Zotero/Better BibTeX: do not edit it.
\usepackage{csquotes}
\usepackage[backend=biber, style=apa]{biblatex}
\addbibresource{../../../references.bib}
\renewcommand*{\bibfont}{\small}

% --- Cross-references (hyperref, then cleveref, always last) ---------
\usepackage{xcolor}
% Dark, desaturated tones: readable on screen and still distinguishable
% from body text when printed in grayscale.
\definecolor{linkinternal}{HTML}{1F3A93}  % sections, tables, figures, equations
\definecolor{linkcite}{HTML}{1E6B3A}      % citations
\definecolor{linkurl}{HTML}{8A2D6B}       % URLs and DOIs
\usepackage[
  colorlinks=true,
  linkcolor=linkinternal,
  citecolor=linkcite,
  urlcolor=linkurl,
  filecolor=linkurl,
  breaklinks=true,        % let long URLs and DOIs wrap in narrow columns
  bookmarksnumbered=true,
  unicode=true,
  pdfusetitle,            % PDF title and author from \title and \author
  pdfdisplaydoctitle=true,
  pdfstartview=FitH,
  pdflang=en-US
]{hyperref}
\usepackage[capitalise, noabbrev]{cleveref}

% --- Macros ----------------------------------------------------------
\newcommand{\R}{\mathbb{R}}
\newcommand{\E}{\mathbb{E}}
