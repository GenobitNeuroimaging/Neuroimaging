% =====================================================================
% preamble.tex — arXiv-style layout, bibliography and macros.
% Single column, 1in margins, centered title block with rules and an
% indented abstract, following the common arXiv preprint look. It is
% self-contained: no arxiv.sty is needed, only standard packages.
% Bibliography settings mirror docs/manuscript/v1/preamble.tex.
% =====================================================================

% --- Encoding and fonts ----------------------------------------------
\usepackage[T1]{fontenc}
\usepackage[utf8]{inputenc}
\usepackage{lmodern}
\usepackage[american]{babel}
\usepackage{microtype}

% --- Layout ----------------------------------------------------------
\usepackage[margin=1in]{geometry}
\usepackage{setspace}
\setstretch{1.05}
\usepackage[small]{titlesec}
\usepackage{fancyhdr}
\pagestyle{fancy}
\fancyhf{}
\renewcommand{\headrulewidth}{0pt}
\fancyhead[L]{\small\itshape Methodology: Data Handling and Architecture}
\fancyfoot[C]{\thepage}
\fancypagestyle{plain}{\fancyhf{}\fancyfoot[C]{\thepage}\renewcommand{\headrulewidth}{0pt}}

% arXiv-style title block.
\makeatletter
\renewcommand{\maketitle}{%
  \par\begingroup
  \renewcommand{\thefootnote}{\fnsymbol{footnote}}
  \begin{center}
    \rule{\linewidth}{1.2pt}\vspace{0.8em}\par
    {\LARGE\bfseries\@title\par}
    \vspace{0.6em}\rule{\linewidth}{0.4pt}\vspace{1.2em}\par
    {\large\begin{tabular}[t]{c@{\hspace{2.5em}}c}\@author\end{tabular}\par}
    \vspace{0.8em}{\normalsize\@date\par}
  \end{center}
  \endgroup\par\vspace{1em}
  \setcounter{footnote}{0}
}
\makeatother

% Abstract: indented, small, centered heading (as in arxiv.sty).
\renewenvironment{abstract}{%
  \begin{center}\bfseries\abstractname\end{center}\vspace{-0.5em}%
  \begin{quote}\small}{\end{quote}\vspace{1em}}

% --- Math, figures and tables ----------------------------------------
\usepackage{amsmath, amssymb}
\usepackage{graphicx}
\graphicspath{{figures/}}
\usepackage{array}
\usepackage{booktabs}
\usepackage[font=small, labelfont=bf, labelsep=period]{caption}
\usepackage{subcaption}
\usepackage{enumitem}
\setlist{nosep}

% --- Bibliography (APA) ----------------------------------------------
% Single source of references for the whole repository.
% references.bib is exported by Zotero/Better BibTeX: do not edit it.
\usepackage{csquotes}
\usepackage[backend=biber, style=apa]{biblatex}
\addbibresource{../../references.bib}

% --- Cross-references (hyperref, then cleveref, always last) ---------
\usepackage{xcolor}
\definecolor{linkinternal}{HTML}{1F3A93}
\definecolor{linkcite}{HTML}{1E6B3A}
\definecolor{linkurl}{HTML}{8A2D6B}
\usepackage[
  colorlinks=true,
  linkcolor=linkinternal,
  citecolor=linkcite,
  urlcolor=linkurl,
  filecolor=linkurl,
  breaklinks=true,
  bookmarksnumbered=true,
  unicode=true,
  pdfusetitle,
  pdfdisplaydoctitle=true,
  pdfstartview=FitH,
  pdflang=en-US
]{hyperref}
\usepackage[capitalise, noabbrev]{cleveref}

% --- Macros ----------------------------------------------------------
\newcommand{\R}{\mathbb{R}}
\newcommand{\E}{\mathbb{E}}
